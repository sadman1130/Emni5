        // --- GAME STATE ---

            slap: { name: 'SLAP!', dmg: 5, pts: 50, sound: 'slap', anim: 'anim-slap', color: '#f59e0b' },
            punch: { name: 'POW!', dmg: 12, pts: 120, sound: 'punch', anim: 'anim-punch', color: '#ef4444' },

            pie: { name: 'SPLAT!', dmg: 8, pts: 80, sound: 'splat', anim: 'anim-slap', color: '#facc15' },
            water: { name: 'SPLASH!', dmg: 10, pts: 100, sound: 'splash', anim: 'anim-punch', color: '#3b82f6' }
        };

        // --- DOM ELEMENTS ---
        const mascotContainer = document.getElementById('mascot-container');
        const mascotSvg = document.getElementById('sadman-svg');

        // Facial Groups

        const eyesHit = document.getElementById('eyes-hit');
        const eyesKO = document.getElementById('eyes-ko');
        const mouthNormal = document.getElementById('mouth-normal');
        const mouthHit = document.getElementById('mouth-hit');
        const mouthKO = document.getElementById('mouth-ko');

        // --- CANVAS PARTICLES ---
        const canvas = document.getElementById('fx-canvas');
        const ctx = canvas.getContext('2d');
        let particles = [];

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        class Particle {
            constructor(x, y, color) {
                this.x = x;
                this.y = y;
                this.color = color;
                this.size = Math.random() * 8 + 4;
                this.vx = (Math.random() - 0.5) * 12;
                this.vy = (Math.random() - 0.5) * 12 - 2;
                this.gravity = 0.3;
                this.alpha = 1;
                this.decay = Math.random() * 0.03 + 0.015;
            }

            update() {
                this.x += this.vx;
                this.y += this.vy;
                this.vy += this.gravity;
                this.alpha -= this.decay;
            }

            draw() {
                ctx.save();
                ctx.globalAlpha = Math.max(0, this.alpha);
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }
        }

        function spawnParticles(x, y, color, count = 15) {
            for (let i = 0; i < count; i++) {
                particles.push(new Particle(x, y, color));
            }
        }

        function renderParticles() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            for (let i = particles.length - 1; i >= 0; i--) {
                const p = particles[i];
                p.update();
                p.draw();
                if (p.alpha <= 0) particles.splice(i, 1);
            }
            requestAnimationFrame(renderParticles);
        }
        requestAnimationFrame(renderParticles);

        // --- WEB AUDIO API SYNTHESIS (No External Sound Files) ---
        let audioCtx = null;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
        }

        function playSynthesizedSound(type) {
            if (state.soundMuted) return;
            initAudio();
            if (!audioCtx) return;

            const now = audioCtx.currentTime;

            if (type === 'slap') {
                // High crisp slap / noise burst + quick pitch down
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(350, now);
                osc.frequency.exponentialRampToValueAtTime(40, now + 0.1);

                gain.gain.setValueAtTime(0.8, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.1);

                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start(now);
                osc.stop(now + 0.1);
            } 
            else if (type === 'punch') {
                // Heavy bass thump
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(180, now);
                osc.frequency.exponentialRampToValueAtTime(30, now + 0.2);

                gain.gain.setValueAtTime(1.0, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);

                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start(now);
                osc.stop(now + 0.2);
            } 
            else if (type === 'hammer') {
                // Cartoon squeak! High pitch glide
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(400, now);
                osc.frequency.linearRampToValueAtTime(1200, now + 0.08);
                osc.frequency.linearRampToValueAtTime(600, now + 0.18);

                gain.gain.setValueAtTime(0.7, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);

                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start(now);
                osc.stop(now + 0.2);
            } 
            else if (type === 'splat') {
                // Cream pie squishy splat sound
                const bufferSize = audioCtx.sampleRate * 0.15;
                const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
                const data = buffer.getChannelData(0);
                for (let i = 0; i < bufferSize; i++) {
                    data[i] = Math.random() * 2 - 1;
                }
                const noise = audioCtx.createBufferSource();
                noise.buffer = buffer;

                const filter = audioCtx.createBiquadFilter();
                filter.type = 'lowpass';
                filter.frequency.setValueAtTime(800, now);
                filter.frequency.linearRampToValueAtTime(200, now + 0.15);

                const gain = audioCtx.createGain();
                gain.gain.setValueAtTime(0.6, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.15);

                noise.connect(filter);
                filter.connect(gain);
                gain.connect(audioCtx.destination);
                noise.start(now);
            } 
            else if (type === 'splash') {
                // Water splash sound
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(600, now);
                osc.frequency.exponentialRampToValueAtTime(150, now + 0.25);

                gain.gain.setValueAtTime(0.7, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.25);

                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start(now);
                osc.stop(now + 0.25);
            }
            else if (type === 'ko') {
                // Sad slide down whistle on KO
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(500, now);
                osc.frequency.exponentialRampToValueAtTime(80, now + 0.6);

                gain.gain.setValueAtTime(0.5, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.6);

                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start(now);
                osc.stop(now + 0.6);
            }
        }

        // --- GAME ACTIONS & INTERACTION ---

        function handleHit(e) {
            initAudio();

            // Calculate Touch / Mouse Position
            const clientX = e.touches ? e.touches[0].clientX : e.clientX;
            const clientY = e.touches ? e.touches[0].clientY : e.clientY;

            const currentWpn = weapons[state.currentWeapon];

            // Auto Revive if KO'd and hit again
            if (state.isKO) {
                resetGame();
            }

            // Reduce HP
            state.hp = Math.max(0, state.hp - currentWpn.dmg);

            // Combo Calculation
            state.combo++;
            clearTimeout(state.comboTimer);
            state.comboTimer = setTimeout(() => {
                state.combo = 0;
                comboVal.textContent = '0x';
            }, 1200);

            // Score Calculation
            const addedScore = currentWpn.pts * (1 + Math.floor(state.combo / 5) * 0.5);
            state.score += addedScore;

            // Update UI Stats
            scoreVal.textContent = state.score;
            comboVal.textContent = `${state.combo}x`;
            updateHpBar();

            // Trigger Animation on Character
            mascotSvg.classList.remove('anim-slap', 'anim-punch', 'anim-hammer');
            void mascotSvg.offsetWidth; // Force Reflow
            mascotSvg.classList.add(currentWpn.anim);

            // Play Synth Sound
            playSynthesizedSound(currentWpn.sound);

            // Visual Effects (Particles & Text)
            spawnParticles(clientX, clientY, currentWpn.color, 12 + state.combo);
            showFloatingText(clientX, clientY, `${currentWpn.name} -${currentWpn.dmg}`);

            // Specific Weapon Overlay Effects
            if (state.currentWeapon === 'pie') {
                pieSplat.classList.remove('hidden');
            }

            // Facial Expression React
            if (state.hp <= 0) {
                triggerKO();
            } else {
                showHurtFace();
            }
        }

        function showHurtFace() {
            eyesNormal.classList.add('hidden');
            eyesHit.classList.remove('hidden');
            mouthNormal.classList.add('hidden');
            mouthHit.classList.remove('hidden');

            // Show bruise on lower HP
            if (state.hp < 60) {
                bruises.classList.remove('opacity-0');
            }

            // Revert face after brief delay
            clearTimeout(mascotSvg.faceTimer);
            mascotSvg.faceTimer = setTimeout(() => {
                if (!state.isKO) {
                    eyesNormal.classList.remove('hidden');
                    eyesHit.classList.add('hidden');
                    mouthNormal.classList.remove('hidden');
                    mouthHit.classList.add('hidden');
                }
            }, 450);
        }

        function triggerKO() {
            state.isKO = true;
            eyesNormal.classList.add('hidden');
            eyesHit.classList.add('hidden');
            eyesKO.classList.remove('hidden');

            mouthNormal.classList.add('hidden');
            mouthHit.classList.add('hidden');
            mouthKO.classList.remove('hidden');

            playSynthesizedSound('ko');
            showFloatingText(window.innerWidth / 2, window.innerHeight / 2 - 50, "KNOCKOUT!", "#ef4444");
        }

        function updateHpBar() {
            const pct = (state.hp / state.maxHp) * 100;
            hpBar.style.width = `${pct}%`;
            hpText.textContent = `${state.hp} / ${state.maxHp} HP`;
        }

        function showFloatingText(x, y, text, customColor = null) {
            const el = document.createElement('div');
            el.className = 'floating-text text-2xl md:text-3xl';
            el.textContent = text;
            el.style.left = `${x}px`;
            el.style.top = `${y}px`;
            el.style.color = customColor || weapons[state.currentWeapon].color;

            document.body.appendChild(el);

            setTimeout(() => {
                el.remove();
            }, 800);
        }

        function resetGame() {
            state.hp = state.maxHp;
            state.isKO = false;
            updateHpBar();

            // Reset Face
            eyesNormal.classList.remove('hidden');
            eyesHit.classList.add('hidden');
            eyesKO.classList.add('hidden');

            mouthNormal.classList.remove('hidden');
            mouthHit.classList.add('hidden');
            mouthKO.classList.add('hidden');

            pieSplat.classList.add('hidden');
            bruises.classList.add('opacity-0');
        }

        // --- EVENT LISTENERS ---

        // Weapon Selector
        weaponBtns.forEach(btn => {
            btn.addEventListener('click', (e) => {
                weaponBtns.forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                state.currentWeapon = btn.dataset.weapon;
            });
        });

        // Click / Touch Hits
        mascotContainer.addEventListener('mousedown', (e) => {
            handleHit(e);
        });

        mascotContainer.addEventListener('touchstart', (e) => {
            e.preventDefault(); // Prevent double tap zoom / scrolling
            handleHit(e);
        }, { passive: false });

        // Mute Toggle
        muteBtn.addEventListener('click', () => {
            state.soundMuted = !state.soundMuted;
            muteBtn.textContent = state.soundMuted ? '🔇' : '🔊';
        });

        // Reset Button
        resetBtn.addEventListener('click', () => {
            state.score = 0;
            state.combo = 0;
            scoreVal.textContent = '0';
            comboVal.textContent = '0x';
            resetGame();
        });

    </script>
</body>
